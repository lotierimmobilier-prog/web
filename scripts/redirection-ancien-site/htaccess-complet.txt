# ==== DÉBUT : redirection de lotier-immobilier.com vers lotier.immo ====
# Pour revenir en arrière : supprimer ce bloc (jusqu'à la ligne « FIN »).
<IfModule mod_rewrite.c>
RewriteEngine On

# Ce qui continue de fonctionner sur l'ancien site : les sous-domaines, l'administration WordPress, les photos,
# la synchronisation des annonces (ImmoWP), le robot du nouveau site (LotierV2-sync) et un administrateur connecté.
RewriteCond %{HTTP_HOST} !^(www\.)?lotier-immobilier\.com$ [NC,OR]
RewriteCond %{THE_REQUEST} \s/(wp-admin|wp-includes|wp-content|wp-json)[/?\s] [NC,OR]
RewriteCond %{THE_REQUEST} \s/(wp-login|wp-cron|xmlrpc)\.php[?\s] [NC,OR]
RewriteCond %{HTTP_USER_AGENT} LotierV2-sync [NC,OR]
RewriteCond %{HTTP_COOKIE} wordpress_logged_in_ [NC]
RewriteRule ^ - [E=LOTIER_ANCIEN:1]

# Pour ces demandes-là, WordPress répond comme avant, et rien n'est redirigé.
RewriteCond %{ENV:LOTIER_ANCIEN} =1
RewriteRule ^index\.php$ - [L]
RewriteCond %{ENV:LOTIER_ANCIEN} =1
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteRule . /index.php [L]
RewriteCond %{ENV:LOTIER_ANCIEN} =1
RewriteRule ^ - [L]

# Annonces : même adresse sur le nouveau site

RewriteRule ^ https://lotier.immo/ [R=301,L,QSD]
